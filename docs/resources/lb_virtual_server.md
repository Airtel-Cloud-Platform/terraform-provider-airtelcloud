# airtelcloud_lb_virtual_server (Resource)

Manages a Virtual Server on an Airtel Cloud Load Balancer Service. A virtual server defines L4/L7 load balancing rules with backend nodes.

## Example Usage

### HTTP Virtual Server

```terraform
resource "airtelcloud_lb_virtual_server" "http" {
  lb_service_id     = airtelcloud_lb_service.example.id
  name              = "http-vs"
  vip_port_id       = airtelcloud_lb_vip.example.id
  protocol          = "HTTP"
  port              = 80
  routing_algorithm = "ROUND_ROBIN"
  vpc_id            = "029ac9b8-d93e-4691-a7cb-2f651c607cfe"
  interval          = 30
  x_forwarded_for   = true

  nodes = [
    {
      compute_name = "web-server-1"
      compute_ip   = "10.1.1.10"
      port         = 8080
      weight       = 10
    },
    {
      compute_name = "web-server-2"
      compute_ip   = "10.1.1.11"
      port         = 8080
      weight       = 10
    },
  ]

  timeouts {
    create = "15m"
  }
}
```

### HTTPS Virtual Server with SSL Termination

```terraform
resource "airtelcloud_lb_virtual_server" "https" {
  lb_service_id     = airtelcloud_lb_service.example.id
  name              = "https-vs"
  vip_port_id       = airtelcloud_lb_vip.example.id
  protocol          = "HTTPS"
  port              = 443
  routing_algorithm = "LEAST_CONNECTIONS"
  vpc_id            = "029ac9b8-d93e-4691-a7cb-2f651c607cfe"
  interval          = 30
  certificate_id    = airtelcloud_lb_certificate.example.id

  persistence_enabled = true
  persistence_type    = "cookie"

  nodes = [
    {
      compute_name = "web-server-1"
      compute_ip   = "10.1.1.10"
      port         = 8443
      weight       = 10
    },
  ]
}
```
